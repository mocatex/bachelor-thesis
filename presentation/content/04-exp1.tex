\begin{frame}{Lesson \#1: the advantage that wasn't}{How an apparent quantum win dissolved}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{alertblock}{The early result}
                A first run showed a \emph{large} quantum lead, the classical
                mirror used a \textbf{ReLU} block that collapsed onto the majority class.
            \end{alertblock}
            \begin{exampleblock}{After a fair rebuild}
                Swapping the mirror's activation to \textbf{$\tanh$} closed the gap
                entirely. The two models tied.
            \end{exampleblock}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{callout}[Takeaway]
                The ``advantage'' lived in an \textbf{under-built baseline}, not in
                the quantum circuit.
                \vspace{4pt}

                Spiky validation curves looked like a barren plateau, but they
                appear \emph{just as strongly} in the classical mirror, so they are
                model-selection noise, not a quantum pathology.
            \end{callout}
        \end{column}
    \end{columns}
\end{frame}

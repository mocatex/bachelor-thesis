\begin{frame}{Setup: a quantum filter slid across the signal}{Does unitarity buy noise robustness?}
    \centering

    \vspace{2pt}
    {\scriptsize\color{zhawgrey}\textbf{Figure:}~pool $\to$ patch $\to$ spherical

        $\to$ shared deep head.}
    \vspace{3pt}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{block}{The hypothesis}
                Unitary dynamics $\Rightarrow$ a restricted Lipschitz
                constant~\cite{sokolic2017robust} $\Rightarrow$ intrinsic resistance to
                perturbations.
            \end{block}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{block}{Strict parity}
                \textbf{52} quantum params vs.\ \textbf{46} classical. Tested under
                OOD \& adversarial noise, with a heavy and a light head.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
